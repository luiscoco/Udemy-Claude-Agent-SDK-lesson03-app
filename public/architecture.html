13. Show before/after and verified outcome

Source: src/client.ts · lines 11–20

function showWorkspace(data: any) {
  const id = data.stage === "after" ? "after" : "before";
  $(id).textContent = data.source; Prism.highlightElement($(id));
  $(id + "-checks").textContent = `${data.checks.passed}/${data.checks.total} checks passed`;
  if (id === "after") {
    $("verdict").textContent = data.checks.available && data.checks.passed === data.checks.total && data.changed ? "Verified fix" : "Not verified";
    $("checks").replaceChildren();
    for (const item of data.checks.cases) { const li = document.createElement("li"); li.textContent = `${item.passed ? "PASS" : "FAIL"} · ${item.label}`; $("checks").append(li); }
  }
}

How this code explains step 13

The browser renders actual source with Python syntax colours and derives Verified fix only when the source changed and all independent checks passed.
